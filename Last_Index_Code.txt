    <script>
        var StorageKey = "TruckFlow_Assignments";
        var HistoryKey = "TruckFlow_History";

        function GetAssignments() {
            var DataV = localStorage.getItem(StorageKey);
            if (!DataV) return [];
            try {
                var Parsed = JSON.parse(DataV);
                return Array.isArray(Parsed) ? Parsed : [];
            } catch (e) { return []; }
        }

        function SaveAssignments(AssignmentsV) {
            localStorage.setItem(StorageKey, JSON.stringify(AssignmentsV));
        }

        function GetHistory() {
            var DataV = localStorage.getItem(HistoryKey);
            if (!DataV) return [];
            try {
                var Parsed = JSON.parse(DataV);
                return Array.isArray(Parsed) ? Parsed : [];
            } catch (e) { return []; }
        }

        function SaveHistory(HistoryV) {
            localStorage.setItem(HistoryKey, JSON.stringify(HistoryV));
        }

        function CreateRampOptions() {
            var XRampSelect = document.getElementById("xRampSelect");
            var DRampSelect = document.getElementById("dRampSelect");
            if (!XRampSelect || !DRampSelect) return;

            for (var i = 1; i <= 40; i++) {
                var rNum = String(i).padStart(2, "0");
                var rName = "X-" + rNum;
                var opt = document.createElement("option");
                opt.value = rName;
                opt.textContent = rName;
                XRampSelect.appendChild(opt);
            }

            for (var j = 1; j <= 40; j++) {
                var dNum = String(j).padStart(2, "0");
                var dName = "D-" + dNum;
                var opt2 = document.createElement("option");
                opt2.value = dName;
                opt2.textContent = dName;
                DRampSelect.appendChild(opt2);
            }
        }

        function ShowMessage(ElementIdV, MessageV) {
            var El = document.getElementById(ElementIdV);
            if (!El) return;
            El.textContent = MessageV;
            setTimeout(function() { El.textContent = ""; }, 3000);
        }

        function AssignRamp(SitePrefixV) {
            var RampSelectV, TruckInputV, PlateInputV, SlotInputV, OperationSelectV, MessageIdV;

            if (SitePrefixV === "X") {
                RampSelectV = document.getElementById("xRampSelect");
                TruckInputV = document.getElementById("xTruckInput");
                PlateInputV = document.getElementById("xPlateInput");
                SlotInputV = document.getElementById("xSlotInput");
                OperationSelectV = document.getElementById("xOperationSelect");
                MessageIdV = "xMessage";
            } else {
                RampSelectV = document.getElementById("dRampSelect");
                TruckInputV = document.getElementById("dTruckInput");
                PlateInputV = document.getElementById("dPlateInput");
                SlotInputV = document.getElementById("dSlotInput");
                OperationSelectV = document.getElementById("dOperationSelect");
                MessageIdV = "dMessage";
            }

            var RampV = RampSelectV.value.trim();
            var TruckV = TruckInputV.value.trim();
            var PlateV = PlateInputV.value.trim();
            var SlotV = SlotInputV.value.trim();
            var OperationV = OperationSelectV.value;

            // Kötelező mezők validálása
            if (RampV === "") {
                ShowMessage(MessageIdV, "A rámpa kiválasztása kötelező!");
                return;
            }
            if (TruckV === "") {
                ShowMessage(MessageIdV, "A kamion / útvonal megadása kötelező!");
                return;
            }
            if (OperationV === "") {
                ShowMessage(MessageIdV, "A művelet kiválasztása kötelező!");
                return;
            }

            var AssignmentsV = GetAssignments();
            var ExistingV = AssignmentsV.find(function(item) { return item.ramp === RampV; });
            if (ExistingV) {
                ShowMessage(MessageIdV, "Ez a rámpa már foglalt!");
                return;
            }

            var NewAssignmentV = {
                ramp: RampV,
                truck: TruckV,
                plate: PlateV,
                slot: SlotV,
                operation: OperationV,
                time: new Date().toLocaleTimeString("hu-HU", { hour: "2-digit", minute: "2-digit" })
            };

            AssignmentsV.push(NewAssignmentV);
            SaveAssignments(AssignmentsV);

            // Mezők alaphelyzetbe állítása
            RampSelectV.value = "";
            TruckInputV.value = "";
            PlateInputV.value = "";
            SlotInputV.value = "";
            OperationSelectV.value = "";

            RenderRamps();
            ShowMessage(MessageIdV, "Sikeres hozzárendelés.");
        }

        function DeleteAssignment(RampV) {
            var ConfirmV = confirm("Biztosan felszabadítod a(z) " + RampV + " rámpát? (Az autó átkerül az előzményekbe)");
            if (!ConfirmV) return;

            var AssignmentsV = GetAssignments();
            var HistoryV = GetHistory();

            var targetItem = AssignmentsV.find(function(item) { return item.ramp === RampV; });
            if (targetItem) {
                targetItem.leftTime = new Date().toLocaleTimeString("hu-HU", { hour: "2-digit", minute: "2-digit" });
                HistoryV.unshift(targetItem);
                SaveHistory(HistoryV);
            }

            AssignmentsV = AssignmentsV.filter(function(item) { return item.ramp !== RampV; });
            SaveAssignments(AssignmentsV);
            RenderRamps();
        }

        function CreateRampRow(TableElementV, RampNameV, AssignmentsV) {
            var AssignmentV = AssignmentsV.find(function(ItemV) { return ItemV.ramp === RampNameV; });

            var RowV = document.createElement("tr");
            var RampCellV = document.createElement("td");
            var TruckCellV = document.createElement("td");
            var PlateCellV = document.createElement("td");
            var SlotCellV = document.createElement("td");
            var OperationCellV = document.createElement("td");
            var TimeCellV = document.createElement("td");
            var ButtonCellV = document.createElement("td");

            RampCellV.textContent = RampNameV;

            if (AssignmentV) {
                TruckCellV.textContent = AssignmentV.truck;
                PlateCellV.textContent = AssignmentV.plate || "-";
                SlotCellV.textContent = AssignmentV.slot || "-";
                OperationCellV.textContent = AssignmentV.operation;
                TimeCellV.textContent = AssignmentV.time;
                RowV.className = "occupied";

                var DeleteButtonV = document.createElement("button");
                DeleteButtonV.type = "button";
                DeleteButtonV.textContent = "X";
                DeleteButtonV.className = "delete-btn";
                DeleteButtonV.addEventListener("click", function() {
                    DeleteAssignment(RampNameV);
                });
                ButtonCellV.appendChild(DeleteButtonV);
            } else {
                TruckCellV.textContent = "Szabad";
                PlateCellV.textContent = "-";
                SlotCellV.textContent = "-";
                OperationCellV.textContent = "-";
                TimeCellV.textContent = "-";
                RowV.className = "free";
            }

            RowV.appendChild(RampCellV);
            RowV.appendChild(TruckCellV);
            RowV.appendChild(PlateCellV);
            RowV.appendChild(SlotCellV);
            RowV.appendChild(OperationCellV);
            RowV.appendChild(TimeCellV);
            RowV.appendChild(ButtonCellV);

            TableElementV.appendChild(RowV);
        }

        function RenderRamps() {
            var XTableV = document.getElementById("xRampTable");
            var DTableV = document.getElementById("dRampTable");
            if (!XTableV || !DTableV) return;

            XTableV.innerHTML = "";
            DTableV.innerHTML = "";

            var AssignmentsV = GetAssignments();

            for (var i = 1; i <= 40; i++) {
                CreateRampRow(XTableV, "X-" + String(i).padStart(2, "0"), AssignmentsV);
            }
            for (var j = 1; j <= 40; j++) {
                CreateRampRow(DTableV, "D-" + String(j).padStart(2, "0"), AssignmentsV);
            }
        }

        function RenderHistory() {
            var HistoryBody = document.getElementById("historyTableBody");
            if (!HistoryBody) return;
            HistoryBody.innerHTML = "";

            var HistoryV = GetHistory();
            if (HistoryV.length === 0) {
                var tr = document.createElement("tr");
                var td = document.createElement("td");
                td.colSpan = 7;
                td.textContent = "Még nincsenek előzmények.";
                td.style.textAlign = "center";
                td.style.padding = "15px";
                tr.appendChild(td);
                HistoryBody.appendChild(tr);
                return;
            }

            HistoryV.forEach(function(item) {
                var tr = document.createElement("tr");
                
                var rCell = document.createElement("td");
                rCell.textContent = item.ramp;
                
                var tCell = document.createElement("td");
                tCell.textContent = item.truck;
                tCell.style.fontWeight = "bold";

                var pCell = document.createElement("td");
                pCell.textContent = item.plate || "-";

                var sCell = document.createElement("td");
                sCell.textContent = item.slot || "-";
                
                var oCell = document.createElement("td");
                oCell.textContent = item.operation;
                
                var timeCell = document.createElement("td");
                timeCell.textContent = item.time;

                var leftCell = document.createElement("td");
                leftCell.textContent = item.leftTime || "-";

                tr.appendChild(rCell);
                tr.appendChild(tCell);
                tr.appendChild(pCell);
                tr.appendChild(sCell);
                tr.appendChild(oCell);
                tr.appendChild(timeCell);
                tr.appendChild(leftCell);

                HistoryBody.appendChild(tr);
            });
        }

        document.addEventListener("DOMContentLoaded", function() {
            document.getElementById("xAssignBtn").addEventListener("click", function() { AssignRamp("X"); });
            document.getElementById("dAssignBtn").addEventListener("click", function() { AssignRamp("D"); });
            document.getElementById("xRefreshBtn").addEventListener("click", RenderRamps);
            document.getElementById("dRefreshBtn").addEventListener("click", RenderRamps);

            var modal = document.getElementById("historyModal");
            document.getElementById("openHistoryBtn").addEventListener("click", function() {
                RenderHistory();
                modal.style.display = "flex";
            });
            document.getElementById("closeHistoryBtn").addEventListener("click", function() {
                modal.style.display = "none";
            });

            document.getElementById("clearHistoryBtn").addEventListener("click", function() {
                var password = prompt("Add meg a jelszót az előzmények törléséhez:");
                if (password === null) return;

                if (password === "aaa") {
                    var confirmClear = confirm("Biztosan törölni akarod az összes előzményt?");
                    if (confirmClear) {
                        localStorage.removeItem(HistoryKey);
                        RenderHistory();
                        alert("Az előzmények sikeresen törölve lettek.");
                    }
                } else {
                    alert("Hibás jelszó!");
                }
            });

            CreateRampOptions();
            RenderRamps();

            window.addEventListener("storage", function(event) {
                if (event.key === StorageKey) {
                    RenderRamps();
                }
            });
        });
